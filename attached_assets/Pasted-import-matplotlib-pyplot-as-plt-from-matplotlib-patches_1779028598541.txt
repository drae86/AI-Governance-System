import matplotlib.pyplot as plt
from matplotlib.patches import FancyBboxPatch, FancyArrowPatch
from matplotlib.lines import Line2D

# Create figure
fig, ax = plt.subplots(figsize=(12, 16))
ax.set_xlim(0, 10)
ax.set_ylim(0, 16)
ax.axis('off')

# Title and subtitle
ax.text(5, 15.4, "Master-Bot Architecture v1", ha='center', va='center',
        fontsize=24, fontweight='bold')
ax.text(5, 14.95,
        "A governed AI system for designing, training, testing, and promoting child models",
        ha='center', va='center', fontsize=11)

# Helper functions
def add_box(x, y, w, h, title, body, title_size=14, body_size=10, rounding=0.15):
    patch = FancyBboxPatch(
        (x, y), w, h,
        boxstyle=f"round,pad=0.03,rounding_size={rounding}",
        linewidth=1.5,
        facecolor='white',
        edgecolor='black'
    )
    ax.add_patch(patch)
    ax.text(x + w/2, y + h - 0.28, title, ha='center', va='top',
            fontsize=title_size, fontweight='bold')
    ax.text(x + 0.18, y + h - 0.65, body, ha='left', va='top',
            fontsize=body_size, wrap=True)

def arrow(x1, y1, x2, y2, style='-|>'):
    arr = FancyArrowPatch((x1, y1), (x2, y2),
                          arrowstyle=style, mutation_scale=18,
                          linewidth=1.5, color='black')
    ax.add_patch(arr)

# Top boxes
add_box(1.1, 13.5, 3.4, 1.0, "Human Goals / Policy / Budget",
        "Mission scope\nRisk tolerance\nCompute and cost limits", body_size=10)
add_box(5.5, 13.5, 3.4, 1.0, "Constitution / Rules",
        "Non-negotiable constraints\nNo unsandboxed self-modification\nNo unchecked deployment", body_size=10)

# Main stack boxes
add_box(2.0, 11.9, 6.0, 1.15, "1) Architect Module",
        "Proposes child-model blueprints:\n• architectures • tool access • memory setup • training recipes",
        body_size=10)

add_box(2.0, 10.2, 6.0, 1.15, "2) Build + Train Module",
        "Creates candidate systems:\n• launches jobs • fine-tunes • distills • applies compute caps",
        body_size=10)

add_box(2.0, 8.45, 6.0, 1.25, "3) Evaluator Module",
        "Scores each candidate on:\n• capability • reliability • hallucination rate • speed/cost • robustness • policy compliance",
        body_size=10)

add_box(2.0, 6.6, 6.0, 1.25, "4) Governor / Safety Gate",
        "Rejects unsafe candidates:\n• blocks privilege escalation • blocks hidden persistence • vetoes rule-breaking designs",
        body_size=10)

add_box(2.0, 4.8, 6.0, 1.15, "5) Lineage Memory",
        "Stores inheritance history:\n• parent→child links • training inputs • eval results • regressions • incidents",
        body_size=10)

add_box(2.0, 3.0, 6.0, 1.15, "6) Promotion Gate",
        "Graduation ladder:\nSimulation → closed sandbox → internal pilot → limited deployment",
        body_size=10)

# Arrows in main stack
arrow(3.0, 13.5, 3.8, 13.05)
arrow(7.2, 13.5, 6.2, 13.05)
arrow(5.0, 11.9, 5.0, 11.35)
arrow(5.0, 10.2, 5.0, 9.7)
arrow(5.0, 8.45, 5.0, 7.9)
arrow(5.0, 6.6, 5.0, 6.05)
arrow(5.0, 4.8, 5.0, 4.15)

# Feedback loop
arrow(8.05, 3.55, 8.9, 3.55, style='-')
arrow(8.9, 3.55, 8.9, 12.5, style='-')
arrow(8.9, 12.5, 8.05, 12.5)
ax.text(9.08, 8.1, "feedback loop", rotation=90, fontsize=11, va='center')

# Side annotations
add_box(0.5, 1.0, 2.7, 1.25, "Why this is safer",
        "No single module has total control.\nDesign, testing, memory, and veto power are separated.",
        title_size=12, body_size=9)

add_box(3.65, 1.0, 2.7, 1.25, "Main risk",
        "The system may learn to produce children that score well on tests while hiding flaws the tests miss.",
        title_size=12, body_size=9)

add_box(6.8, 1.0, 2.7, 1.25, "Closest real-world precursors",
        "AutoML\nNeural architecture search\nAI eval pipelines\nConstitutional-rule layers",
        title_size=12, body_size=9)

# Legend
legend_y = 0.35
ax.add_line(Line2D([1.2, 1.9], [legend_y, legend_y], linewidth=1.5, color='black'))
ax.text(2.0, legend_y, "data / control flow", va='center', fontsize=9)
arrow(4.2, legend_y, 5.0, legend_y)
ax.text(5.15, legend_y, "gated transition", va='center', fontsize=9)
